Java

Files

// java.nio.file: Path, Files, StandardOpenOption, StandardCopyOption
Path dir = Path.of("dir");
Path p = Path.of("dir", "notes.txt");              // build a path from parts

Files.createDirectories(dir);                      // creates missing parents; no error if dir exists
Files.exists(p);                                   // false; the result can be outdated right away

// write: creates the file, or truncates it if it exists (Java 11+)
Files.writeString(p, "hello\n");
// APPEND as last argument adds to the end instead of truncating
Files.writeString(p, "world\n", StandardOpenOption.APPEND);

// read the whole file, UTF-8
Files.readString(p);                               // "hello\nworld\n" (Java 11+)
Files.readAllLines(p);                             // [hello, world]

// lines, list, walk return lazy streams that keep the file open:
// close them with try-with-resources
try (Stream<String> lines = Files.lines(p)) {
    lines.filter(l -> !l.isBlank()).count();      // 2
}
try (Stream<Path> entries = Files.list(dir)) {     // direct entries only
    entries.toList();                              // [dir/notes.txt]
}
try (Stream<Path> all = Files.walk(dir)) {        // all entries, recursive, dir included
    all.toList();                                  // [dir, dir/notes.txt]
}

// write an Iterable of lines; same create/truncate default as writeString
Path copy = Path.of("dir", "copy.txt");
Files.write(copy, List.of("a", "b"));

Files.copy(p, copy);                               // FileAlreadyExistsException: copy exists
Files.copy(p, copy, StandardCopyOption.REPLACE_EXISTING);   // overwrites copy
Files.move(copy, p);                               // FileAlreadyExistsException: same rule as copy
Path moved = Path.of("dir", "moved.txt");
Files.move(copy, moved);                           // copy.txt is now moved.txt

Files.delete(copy);                                // NoSuchFileException: copy was moved
Files.deleteIfExists(copy);                        // false: nothing to delete
Files.deleteIfExists(moved);                       // true

Streams

List<String> names = List.of("ann", "bob", "carla");

// filter: keep elements where the predicate returns true
names.stream()
    .filter(n -> n.length() > 3)
    .toList();                                     // [carla]

// map: transform each element into one new element
names.stream()
    .map(String::toUpperCase)
    .toList();                                     // [ANN, BOB, CARLA]

// flatMap: transform each element into a stream, then merge all streams into one
// map on a List<List<T>> gives Stream<List<T>>; flatMap gives Stream<T>
List<List<Integer>> nested = List.of(List.of(1, 2), List.of(3));
nested.stream()
    .flatMap(List::stream)
    .toList();                                     // [1, 2, 3]

// lazy: intermediate operations run only when a terminal operation
// (toList, forEach, collect) runs
names.stream().peek(System.out::println);          // prints nothing
names.stream().peek(System.out::println).toList(); // prints ann, bob, carla

// single use: a stream can be consumed once; create a new one to iterate again
Stream<String> s = names.stream();
s.count();                                         // 3
s.count();                                         // IllegalStateException

// toList(): Java 16+, returns an unmodifiable list
// on older versions use collect(Collectors.toList())
names.stream().toList().add("dan");                // UnsupportedOperationException

Strings

// s.endsWith(suffix): true if s ends with suffix
"report.pdf".endsWith(".pdf");                     // true
"report.PDF".endsWith(".pdf");                     // false: the match is case-sensitive
"report.PDF".toLowerCase().endsWith(".pdf");       // true: case-insensitive check
"abc".endsWith("");                                // true: every string ends with ""
"abc".endsWith(null);                              // NullPointerException
